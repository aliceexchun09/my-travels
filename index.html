   <script type="text/babel">

       const { useState, useEffect, useMemo, useRef } = React;

       // --- 核心航點座標 ---

       const COUNTRY_DATA = {

           "韓國": { x: 830, y: 180, distance: 1480 },

           "日本": { x: 860, y: 190, distance: 2100 },

           "英國": { x: 485, y: 115, distance: 9780 },

           "美國": { x: 200, y: 180, distance: 10500 },

           "法國": { x: 495, y: 140, distance: 9800 },

           "泰國": { x: 785, y: 280, distance: 2500 },

           "澳洲": { x: 850, y: 430, distance: 7200 },

           "新加坡": { x: 795, y: 320, distance: 3200 },

           "德國": { x: 510, y: 125, distance: 9300 },

           "義大利": { x: 520, y: 160, distance: 9600 },

       };

       const TAIWAN_COORD = { x: 815, y: 245 };

       // ==========================================================

       // ⚠️ 這是您需要修改的地方 ⚠️

       // 請將您從 Firebase 控制台複製的內容，貼在下面的大括號中

       // ==========================================================

       const myFirebaseConfig = {

           // apiKey: "您的 API 金鑰",

           // authDomain: "您的專案網域名稱",

           // ...以此類推

       };

       const isConfigured = myFirebaseConfig && myFirebaseConfig.apiKey;

       if (isConfigured && !firebase.apps.length) {

           firebase.initializeApp(myFirebaseConfig);

       }

       const CanvasMap = ({ visitedCountries }) => {

           const canvasRef = useRef(null);

           useEffect(() => {

               const canvas = canvasRef.current;

               if (!canvas) return;

               const ctx = canvas.getContext('2d');

               ctx.clearRect(0, 0, canvas.width, canvas.height);

               ctx.fillStyle = '#EAE0D5'; ctx.globalAlpha = 0.2;

               [[200, 150, 100], [600, 100, 120], [850, 280, 80]].forEach(([cx, cy, r]) => {

                   ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill();

               });

               visitedCountries.forEach(country => {

                   const dest = COUNTRY_DATA[country] || { x: 500, y: 250 };

                   ctx.setLineDash([5, 5]); ctx.strokeStyle = '#D0B8A8'; ctx.lineWidth = 1.5; ctx.globalAlpha = 0.8;

                   ctx.beginPath(); ctx.moveTo(TAIWAN_COORD.x, TAIWAN_COORD.y);

                   const cpX = (TAIWAN_COORD.x + dest.x) / 2;

                   const cpY = Math.min(TAIWAN_COORD.y, dest.y) - 60;

                   ctx.quadraticCurveTo(cpX, cpY, dest.x, dest.y); ctx.stroke();

                   ctx.setLineDash([]); ctx.fillStyle = '#4A4A4A'; ctx.beginPath(); ctx.arc(dest.x, dest.y, 4, 0, Math.PI * 2); ctx.fill();

                   ctx.font = '10px sans-serif'; ctx.fillStyle = '#9A8C98'; ctx.fillText(country, dest.x + 8, dest.y + 4);

               });

               ctx.globalAlpha = 1; ctx.fillStyle = '#2D2D2D'; ctx.beginPath(); ctx.arc(TAIWAN_COORD.x, TAIWAN_COORD.y, 6, 0, Math.PI * 2); ctx.fill();

               ctx.font = 'bold 12px sans-serif'; ctx.fillText('TAIWAN', TAIWAN_COORD.x - 20, TAIWAN_COORD.y + 20);

           }, [visitedCountries]);

           return <canvas ref={canvasRef} width={1000} height={500} className="w-full h-auto" />;

       };

       const App = () => {

           const [view, setView] = useState(isConfigured ? 'main' : 'setup');

           const [user, setUser] = useState(null);

           const [trips, setTrips] = useState([]);

           const [loading, setLoading] = useState(true);

           const [isModalOpen, setIsModalOpen] = useState(false);

           const [formData, setFormData] = useState({ location: '', country: '', startDate: '', endDate: '', note: '', category: '景點' });

           useEffect(() => {

               if (!isConfigured) { setLoading(false); return; }

               const unsubAuth = firebase.auth().onAuthStateChanged(async (u) => {

                   if (u) { setUser(u); } else { try { await firebase.auth().signInAnonymously(); } catch(e) { console.error("Auth failed:", e); } }

               });

               return () => unsubAuth();

           }, []);

           useEffect(() => {

               if (!user) return;

               const unsub = firebase.firestore().collection('users').doc(user.uid).collection('my_trips')

                   .onSnapshot((snap) => {

                       const data = snap.docs.map(d => ({ id: d.id, ...d.data() }));

                       setTrips(data);

                       setLoading(false);

                   });

               return () => unsub();

           }, [user]);

           const visitedCountries = useMemo(() => [...new Set(trips.map(t => t.country))], [trips]);

           const totalKM = useMemo(() => trips.reduce((s, t) => s + (COUNTRY_DATA[t.country]?.distance || 0), 0), [trips]);

           const handleSave = async (e) => {

               e.preventDefault();

               if (!isConfigured) { alert("請先完成代碼中的 Firebase 配置！"); return; }

               const id = Date.now().toString();

               await firebase.firestore().collection('users').doc(user.uid).collection('my_trips').doc(id).set({ ...formData, id });

               setIsModalOpen(false);

           };

           const handleDelete = async (id) => {

               if (window.confirm('確定要刪除這段回憶嗎？')) {

                   await firebase.firestore().collection('users').doc(user.uid).collection('my_trips').doc(id).delete();

               }

           };

           return (

               <div className="min-h-screen">

                   <nav className="bg-white/80 backdrop-blur-md sticky top-0 z-50 border-b border-[#F0EAD6] px-6 py-4 flex justify-between items-center">

                       <div className="flex items-center gap-2 cursor-pointer" onClick={() => setView('main')}>

                           <span className="text-2xl">🌍</span>

                           <h1 className="text-xl font-bold tracking-tight">Voyage Notes</h1>

                       </div>

                       <div className="flex gap-2">

                           <button onClick={() => setView('setup')} className="text-xs font-bold text-[#9A8C98] px-4 py-2 hover:text-[#2D2D2D]">⚙️ 設定指南</button>

                           <button onClick={() => setIsModalOpen(true)} className="bg-[#2D2D2D] text-white px-5 py-2 rounded-full text-xs font-bold shadow-lg">＋ 新增紀錄</button>

                       </div>

                   </nav>

                   {!isConfigured || view === 'setup' ? (

                       <main className="max-w-2xl mx-auto p-8 animate-in slide-in-from-bottom-4 duration-500">

                           <div className="bg-white border border-[#F0EAD6] p-10 rounded-[40px] shadow-sm space-y-8">

                               <h2 className="text-3xl font-bold border-b pb-4">Firebase 串接步驟</h2>

                               {!isConfigured && (

                                   <div className="bg-amber-50 border border-amber-200 p-4 rounded-2xl text-amber-700 text-sm">

                                       ⚠️ 偵測到尚未貼上配置資訊。請完成以下步驟來啟動雲端存檔功能。

                                   </div>

                               )}

                               <div className="space-y-6">

                                   <div className="tutorial-step">

                                       <h4 className="font-bold">1. 獲取 Config 物件</h4>

                                       <p className="text-sm text-[#9A8C98]">Firebase 控制台 > 專案設定 > 您的應用程式 > 複製 {} 裡的內容。</p>

                                   </div>

                                   <div className="tutorial-step">

                                       <h4 className="font-bold">2. 貼上代碼</h4>

                                       <p className="text-sm text-[#9A8C98]">在 index.html 第 98 行貼上資訊並上傳至 GitHub。</p>

                                   </div>

                                   <div className="tutorial-step">

                                       <h4 className="font-bold">3. 開啟 Firestore</h4>

                                       <p className="text-sm text-[#9A8C98]">在 Firebase 左側選單開啟 Cloud Firestore，並以「測試模式」啟動。</p>

                                   </div>

                                   <div className="tutorial-step">

                                       <h4 className="font-bold">4. 開啟匿名登入</h4>

                                       <p className="text-sm text-[#9A8C98]">在 Authentication > Sign-in method 開啟「匿名」登入。</p>

                                   </div>

                               </div>

                               <button onClick={() => setView('main')} className="w-full bg-[#FAF9F6] py-4 rounded-2xl font-bold text-sm border border-[#F0EAD6]">

                                   {isConfigured ? '← 返回我的 App' : '我已完成貼上，點此進入'}

                               </button>

                           </div>

                       </main>

                   ) : (

                       <main className="max-w-6xl mx-auto p-6 space-y-10 animate-in fade-in duration-700">

                           <div className="grid grid-cols-2 md:grid-cols-3 gap-4">

                               <div className="bg-white border border-[#F0EAD6] p-6 rounded-[32px] text-center">

                                   <p className="text-[10px] font-bold text-[#9A8C98] uppercase mb-1">國家總數</p>

                                   <h3 className="text-3xl font-light">{visitedCountries.length}</h3>

                               </div>

                               <div className="bg-white border border-[#F0EAD6] p-6 rounded-[32px] text-center">

                                   <p className="text-[10px] font-bold text-[#9A8C98] uppercase mb-1">飛行里程</p>

                                   <h3 className="text-3xl font-light">{totalKM.toLocaleString()} <span className="text-xs">KM</span></h3>

                               </div>

                               <div className="hidden md:flex bg-[#FAF9F6] border border-[#F0EAD6] p-6 rounded-[32px] items-center justify-center italic text-sm text-[#9A8C98]">

                                   "與雲端同步中，紀錄永不遺失。"

                               </div>

                           </div>

                           <div className="bg-white p-4 border border-[#F0EAD6] rounded-[40px] shadow-sm">

                               <CanvasMap visitedCountries={visitedCountries} />

                           </div>

                           <div className="grid grid-cols-1 md:grid-cols-3 gap-6">

                               {trips.length > 0 ? trips.map(t => (

                                   <div key={t.id} className="bg-white border border-[#F0EAD6] rounded-[36px] p-8 space-y-4 hover:shadow-md transition-all group relative">

                                       <button onClick={() => handleDelete(t.id)} className="absolute top-4 right-4 opacity-0 group-hover:opacity-100 text-red-300 hover:text-red-500 transition-all text-xs font-bold">刪除</button>

                                       <div className="flex justify-between items-start">

                                           <span className="bg-[#FAF9F6] px-3 py-1 rounded-full text-[10px] font-bold">{t.category}</span>

                                           <span className="text-xs text-[#9A8C98]">{t.startDate}</span>

                                       </div>

                                       <h4 className="text-xl font-bold">📍 {t.location}</h4>

                                       <p className="text-sm text-[#6B705C] italic">「{t.note}」</p>

                                   </div>

                               )) : (

                                   <div className="col-span-full py-20 text-center text-[#9A8C98]">

                                       目前還沒有雲端紀錄。點擊右上角新增地標！

                                   </div>

                               )}

                           </div>

                       </main>

                   )}

                   {isModalOpen && (

                       <div className="fixed inset-0 z-[60] bg-black/20 backdrop-blur-sm flex items-center justify-center p-4">

                           <div className="bg-white w-full max-w-md rounded-[48px] p-10 space-y-6 animate-in zoom-in-95 duration-300 shadow-2xl">

                               <div className="flex justify-between items-center">

                                   <h2 className="text-2xl font-bold">新增紀錄</h2>

                                   <button onClick={() => setIsModalOpen(false)} className="text-2xl">✕</button>

                               </div>

                               <div className="space-y-4">

                                   <input placeholder="地點 (如: 首爾)" className="w-full bg-[#FAF9F6] px-6 py-4 rounded-2xl outline-none" onChange={e => setFormData({...formData, location: e.target.value})} />

                                   <input placeholder="國家 (如: 韓國)" className="w-full bg-[#FAF9F6] px-6 py-4 rounded-2xl outline-none" onChange={e => setFormData({...formData, country: e.target.value})} />

                                   <div className="grid grid-cols-2 gap-4">

                                       <input type="date" className="w-full bg-[#FAF9F6] px-4 py-3 rounded-2xl text-xs" onChange={e => setFormData({...formData, startDate: e.target.value})} />

                                       <input type="date" className="w-full bg-[#FAF9F6] px-4 py-3 rounded-2xl text-xs" onChange={e => setFormData({...formData, endDate: e.target.value})} />

                                   </div>

                                   <select className="w-full bg-[#FAF9F6] px-6 py-4 rounded-2xl text-sm" onChange={e => setFormData({...formData, category: e.target.value})}>

                                       <option value="景點">📸 景點參訪</option><option value="美食">🍜 美食饗宴</option><option value="購物">🛍️ 購物戰利品</option>

                                   </select>

                                   <textarea placeholder="寫下心情..." className="w-full bg-[#FAF9F6] px-6 py-4 rounded-2xl h-24 text-sm" onChange={e => setFormData({...formData, note: e.target.value})}></textarea>

                               </div>

                               <button onClick={handleSave} className="w-full bg-[#2D2D2D] text-white py-5 rounded-[24px] font-bold shadow-lg">儲存並同步雲端</button>

                           </div>

                       </div>

                   )}

               </div>

           );

       };

       const root = ReactDOM.createRoot(document.getElementById('root'));

       root.render(<App />);

   </script>

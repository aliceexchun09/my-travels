                }, {});

                chartInstance.current = new Chart(ctx, {

                    type: 'doughnut',

                    data: {

                        labels: Object.keys(counts),

                        datasets: [{

                            data: Object.values(counts),

                            backgroundColor: ['#D0B8A8', '#EAE0D5', '#9A8C98', '#4A4A4A'],

                            borderWidth: 0

                        }]

                    },

                        responsive: true,

                        maintainAspectRatio: false,

                    }

                });
